## 二、SQLite 表结构设计

### 体验层

```sql
-- 体验节点
CREATE TABLE experiences (
  id TEXT PRIMARY KEY,                    -- exp_YYYYMMDD_HHmmss
  L0_text TEXT,                           -- 一句话摘要
  L0_embedding BLOB,                      -- L0的语义向量
  L1_text TEXT,                           -- 关键要点（巩固层补充）
  L2_text TEXT,                           -- 具体细节（巩固层补充）
  L3_raw TEXT,                            -- 完整原文
  emotion_category TEXT,                  -- 情感类别
  emotion_intensity REAL,                 -- 情感强度 0-1
  emotion_valence REAL,                   -- 效价 -1到1
  emotion_arousal REAL,                   -- 唤醒度 0-1
  emotion_target TEXT,                    -- 情感指向的实体ID（可选）
  context_focus TEXT,                      -- 上下文：状态层当前 focus
  context_mood TEXT,                       -- 上下文：状态层当前 mood.label
  context_time_of_day TEXT,                -- 上下文：从时间戳自动推导（凌晨/上午/下午/晚上）
  context_silence_before TEXT,             -- 上下文：距上一条体验的时间差
  context_task TEXT,                       -- 上下文：外部系统可选传入（未来扩展）
  context_extra TEXT,                      -- 上下文：JSON扩展字段（预留）
  importance REAL DEFAULT 0.5,            -- 重要度评分（巩固层计算）
  twist_level TEXT DEFAULT 'none',        -- 扭曲程度标记
  created_at TEXT                         -- ISO时间戳
);

-- 体验层边
CREATE TABLE experience_edges (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  from_id TEXT NOT NULL,                  -- 源体验节点ID
  to_id TEXT NOT NULL,                    -- 目标体验节点ID
  type TEXT NOT NULL,                     -- temporal/thematic/causal/associative
  weight REAL DEFAULT 1.0,               -- 原始权重
  decayed_weight REAL DEFAULT 1.0,        -- 衰减后权重
  emotion_driver TEXT,                    -- 情感驱动（可选）
  access_count INTEGER DEFAULT 0,         -- 被召回次数
  created_at TEXT,
  FOREIGN KEY (from_id) REFERENCES experiences(id),
  FOREIGN KEY (to_id) REFERENCES experiences(id)
);
```

### 信息层

```sql
-- 信息层实体
CREATE TABLE entities (
  id TEXT PRIMARY KEY,                    -- uuid
  type TEXT NOT NULL,                     -- person/place/concept/event/skill/other
  name TEXT NOT NULL,                     -- 实体名
  properties TEXT,                        -- JSON格式的属性字典
  embedding BLOB,                         -- name + properties的语义向量
  emotion_timeline TEXT,                  -- JSON格式的情感时间线（person类型）
  emotion_current TEXT,                   -- JSON格式的当前情感（person类型）
  created_at TEXT,
  updated_at TEXT
);

-- 信息层关系边
CREATE TABLE entity_edges (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  from_id TEXT NOT NULL,
  to_id TEXT NOT NULL,
  relation TEXT NOT NULL,                 -- 自由文本
  embedding BLOB,                         -- relation文本的语义向量
  confidence REAL DEFAULT 0.5,            -- 置信度 0-1
  source TEXT,                            -- 来源体验节点ID
  source_type TEXT,                       -- direct/external/inferred/hearsay/unverified
  verified_at TEXT,
  verify_count INTEGER DEFAULT 0,
  created_at TEXT,
  FOREIGN KEY (from_id) REFERENCES entities(id),
  FOREIGN KEY (to_id) REFERENCES entities(id)
);
```

### 跨层边

```sql
-- 跨层边（体验层 ↔ 信息层）
CREATE TABLE cross_edges (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  from_id TEXT NOT NULL,                  -- 体验节点ID
  to_id TEXT NOT NULL,                    -- 信息层实体ID
  context TEXT,                           -- 关联原因
  weight REAL DEFAULT 1.0,
  created_at TEXT
);
```

### 核心层

```sql
-- 核心层（单行表）
CREATE TABLE core (
  id INTEGER PRIMARY KEY CHECK (id = 1), -- 确保只有一行
  invariant_text TEXT NOT NULL,           -- 不变层文本
  stable_text TEXT NOT NULL,              -- 稳定层文本
  malleable_text TEXT NOT NULL,           -- 可塑层文本
  anchors TEXT,                           -- JSON格式的结构化锚点
  version INTEGER DEFAULT 1,
  last_modified TEXT,
  modification_log TEXT                   -- JSON格式的变更日志
);
```

### 状态层

```sql
-- 状态层（单行表）
CREATE TABLE state (
  id INTEGER PRIMARY KEY CHECK (id = 1), -- 确保只有一行
  mood_valence REAL DEFAULT 0.0,         -- 效价 -1到1
  mood_arousal REAL DEFAULT 0.3,         -- 唤醒度 0-1
  mood_label TEXT DEFAULT '平静',         -- 自然语言标签
  energy_value REAL DEFAULT 0.7,          -- 精力 0-1
  energy_label TEXT DEFAULT '正常',        -- 标签
  focus TEXT,                             -- 注意力（字符串或null）
  confidence_value REAL DEFAULT 0.7,      -- 信心 0-1
  confidence_label TEXT DEFAULT '正常',    -- 标签
  updated_at TEXT
);
```

### 个人信息系统

```sql
-- 个人档案
CREATE TABLE person_profiles (

  interaction_style TEXT,                 -- JSON：相处方式参数
```

## 三、Embedding 模型使用

```python
from sentence_transformers import SentenceTransformer

# 加载模型（首次运行会自动下载，约400MB）
model = SentenceTransformer('BAAI/bge-base-zh')

# 生成向量
vector = model.encode("我和张三讨论了记忆系统")
# 输出：numpy数组，768维

# 存入SQLite时转为BLOB
import numpy as np
blob = vector.tobytes()

# 从SQLite读取时转回
vector = np.frombuffer(blob, dtype=np.float32)
```
