在 `on_ready()` 回调中添加 Bot 实例注入和自动语音频道连接：
```python
@self._bot.event
async def on_ready():
    logger.info(f"Discord Bot 已就绪: {self._bot.user}")
    # 注入 Bot 实例到 VoicePlayer
    voice_player.set_bot(self._bot)
    # 如果配置了语音频道，自动加入
    from backend.config.loader import settings
    if settings.discord.voice_channel_id:
        try:
            success = await voice_player.connect(str(settings.discord.voice_channel_id))
            if success:
                logger.info(f"已自动加入语音频道: {settings.discord.voice_channel_id}")
            else:
                logger.warning(f"自动加入语音频道失败: {settings.discord.voice_channel_id}")
        except Exception as e:
            logger.warning(f"自动加入语音频道异常: {e}")
```

在 `disconnect()` 方法中添加语音频道断开：
```python
async def disconnect(self) -> None:
    """断开 Discord 连接。"""
    # 先断开语音频道
    if voice_player.is_connected():
        await voice_player.disconnect()

    if self._bot and not self._bot.is_closed():
        await self._bot.close()
    if self._task:
        self._task.cancel()
        try:
            await self._task
        except asyncio.CancelledError:
            pass
    self._bot = None
    self._task = None
    logger.info("Discord Bot 已断开")
```

实际上，on_ready 已经处理了自动加入，所以 connect_discord 工具本身不需要额外改动。但需要确保返回信息包含语音频道状态：

```python
async def execute(self, **kwargs) -> Dict:
    from backend.services.agent.sources.discord_source import discord_source
    from backend.services.voice_player import voice_player

    if discord_source is None:
        return {"success": False, "message": "Discord 模块未初始化"}

    if discord_source.is_connected():
        voice_status = "语音频道已连接" if voice_player.is_connected() else "未连接语音频道"
        return {"success": True, "message": f"已连接到 Discord，{voice_status}。"}

    try:
        await discord_source.connect()
        # 等待 Bot 就绪
        import asyncio
        for _ in range(10):
            if discord_source.is_connected():
                break
            await asyncio.sleep(1)

        voice_status = "语音频道已连接" if voice_player.is_connected() else "未连接语音频道"
        return {
            "success": True,
            "message": f"已连接到 Discord，{voice_status}。"
        }
    except Exception as e:
        logger.error(f"Discord 连接失败: {e}")
        return {
            "success": False,
            "message": f"连接 Discord 失败：{e}"
        }
```

        logger.error(f"Discord 断开失败: {e}")

            "success": False,
            "message": f"断开 Discord 失败：{e}"
        }
